    else:

    bg_crop = ori_img[y1:y2, x1:x2].astype(np.float32)
    fg_crop = res_resized.astype(np.float32)

    blended = (1.0 - alpha) * bg_crop + alpha * fg_crop

    out_img[y1:y2, x1:x2] = np.clip(blended, 0, 255).astype(np.uint8)
